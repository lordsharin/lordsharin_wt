<!DOCTYPE html>
<html>
<head>
<title>Calculator</title>
<style>
body { text-align:center; font-family:Arial; }
input, button { padding:10px; margin:5px; }
</style>
</head>
<body>

<h2>Simple Calculator</h2>

<input type="number" id="n1" placeholder="First number">
<input type="number" id="n2" placeholder="Second number">

<br>

<button onclick="calc('+')">+</button>
<button onclick="calc('-')">-</button>
<button onclick="calc('*')">*</button>
<button onclick="calc('/')">/</button>

<h3 id="result">Result:</h3>

<script>
function calc(op) {
    let a = Number(document.getElementById("n1").value);
    let b = Number(document.getElementById("n2").value);
    let r;

    if(op == '+') r = a + b;
    else if(op == '-') r = a - b;
    else if(op == '*') r = a * b;
    else if(op == '/') {
        if(b == 0) {
            document.getElementById("result").innerHTML = "Cannot divide by zero";
            return;
        }
        r = a / b;
    }

    document.getElementById("result").innerHTML = "Result: " + r;
}
</script>

</body>
</html>

Filename: grade.html


<!DOCTYPE html>
<html>
<head>
<title>Grade Calculator</title>
</head>
<body>

<h2>Student Grade Calculator</h2>

<button onclick="calculate()">Calculate Grade</button>

<script>
function calculate() {
    let m1 = Number(prompt("Enter subject 1 marks"));
    let m2 = Number(prompt("Enter subject 2 marks"));
    let m3 = Number(prompt("Enter subject 3 marks"));

    let total = m1 + m2 + m3;
    let per = total / 3;
    let grade;

    if(per >= 90) grade = "A";
    else if(per >= 75) grade = "B";
    else if(per >= 60) grade = "C";
    else if(per >= 40) grade = "D";
    else grade = "Fail";

    alert("Total: " + total +
          "\nPercentage: " + per.toFixed(2) +
          "%\nGrade: " + grade);
}
</script>

</html>



3.Age Calculator
Filename: age.html


<!DOCTYPE html>
<html>
<head>
<title>Age Calculator</title>
</head>
<body>

<h2>Age Calculator</h2>

Enter Date of Birth:
<input type="date" id="dob">

<button onclick="calculateAge()">Calculate Age</button>

<h3 id="result"></h3>

<script>
function calculateAge() {
    let dob = new Date(document.getElementById("dob").value);
    let today = new Date();

    if(!document.getElementById("dob").value || dob > today) {
        document.getElementById("result").innerHTML = "Enter a valid date";
        return;
    }

    let age = today.getFullYear() - dob.getFullYear();

    if(today.getMonth() < dob.getMonth() ||
      (today.getMonth() == dob.getMonth() &&
       today.getDate() < dob.getDate())) {
        age--;
    }

    document.getElementById("result").innerHTML =
    "Your Age is: " + age + " years";
}
</script>

</body>
</html>

    let task = document.getElementById("task").value;

    if(task == "") {
        alert("Enter a task");
        return;

    li.innerHTML = task;

9.Number Checking Application
Filename: number.html


<!DOCTYPE html>
<html>
<head>
<title>Number Checking</title>
</head>
<body>

<h2>Number Checking Application</h2>

<input type="number" id="num" placeholder="Enter a number">

<button onclick="check()">Check Number</button>

<h3 id="result"></h3>

<script>
function check() {
    let n = Number(document.getElementById("num").value);
    let r = "";

    if(document.getElementById("num").value == "") {
        document.getElementById("result").innerHTML = "Enter a number";
        return;
    }

    if(n > 0) r += "Positive<br>";
    else if(n < 0) r += "Negative<br>";
    else r += "Zero<br>";

    if(n % 2 == 0) r += "Even<br>";
    else r += "Odd<br>";

    let prime = true;

    if(n <= 1) prime = false;

    for(let i = 2; i <= Math.sqrt(n); i++) {
        if(n % i == 0) {
            prime = false;
            break;
        }
    }

    if(prime) r += "Prime Number";
    else r += "Not Prime Number";

    document.getElementById("result").innerHTML = r;
}
</script>

</body>
</html>
