3.Employee Management System (Node.js + MySQL)


Step 1: Install packages
Open the terminal in your project folder:
npm init -y
npm install express mysql2


Step 2: Create database and table in MySQL
CREATE DATABASE company;
USE company;

CREATE TABLE employees (
    id INT PRIMARY KEY,
    name VARCHAR(50),
    salary INT,
    department VARCHAR(50)
);


Step 3: Create employee.js
const express = require('express');
const mysql = require('mysql2');



code:

const app = express();
app.use(express.json());

const db = mysql.createConnection({
    host: 'localhost',
    user: 'root',
    password: 'your_password',
    database: 'company'
});

db.connect(err => {
    if (err) throw err;
    console.log("MySQL Connected");
});

// Insert employee
app.post('/add', (req, res) => {
    const e = req.body;
    db.query("INSERT INTO employees VALUES (?, ?, ?, ?)",
    [e.id, e.name, e.salary, e.department],
    () => res.send("Employee Added"));
});

// Display employees
app.get('/employees', (req, res) => {
    db.query("SELECT * FROM employees",
    (err, result) => res.json(result));
});

// Update employee
app.put('/update/:id', (req, res) => {
    db.query("UPDATE employees SET salary=? WHERE id=?",
    [req.body.salary, req.params.id],
    () => res.send("Employee Updated"));
});

// Delete employee
app.delete('/delete/:id', (req, res) => {
    db.query("DELETE FROM employees WHERE id=?",
    [req.params.id],
    () => res.send("Employee Deleted"));
});

app.listen(3000, () => console.log("Server running"));


Replace your_password with your actual MySQL password. If your MySQL account has no password, use an empty string ('').
Step 4: Run the program
node employee.js
